# Japanese Vocabulary Trainer — Learned / Review settings
#
# This file controls the simple thresholds used for directional learning status.
# It does NOT change vocabulary.csv and does NOT change existing statistics.
#
# Each direction is evaluated independently:
#   Japanese -> English
#   English -> Japanese
#
# A direction is learned when ALL three learned criteria are met:
#   1. minimum number of attempts
#   2. minimum accuracy
#   3. minimum number of correct answers
#
# Fully learned is NOT a configurable setting.
# It is derived by the application:
#   both directions learned  = fully learned
#   exactly one direction    = partially learned
#   neither direction        = not learned
#
# Needs Review is also evaluated separately per direction.
# It is independent from Learned: a learned direction can still need review.
#
# A direction needs review when either:
#   - wrong answers since the last correct answer reach wrong_answers_to_review, OR
#   - the configured number of days has passed since the last correct answer.
#
# A correct answer resets the wrong-answer review counter for that direction.
# A word does NOT stop being learned just because it needs review.
#
# ---------------- CURRENT SETTINGS ----------------
min_attempts=10
min_accuracy=90
min_correct=5
review_days=30
wrong_answers_to_review=1
review_recovery_accuracy=80
review_recovery_min_answers=5

# ---------------- SETTING REFERENCE ----------------
# min_attempts
#   Minimum total attempts required before a direction can be learned.
#
# min_accuracy
#   Minimum percentage of correct answers required.
#   Example: 90 means at least 90% accuracy.
#
# min_correct
#   Minimum number of correct answers required.
#
# review_days
#   Number of days since the last correct answer after which the direction needs review.
#
# wrong_answers_to_review
#   Number of wrong answers since the last correct answer that immediately triggers review.
#   The default value 1 means one wrong answer is enough.
#
# review_recovery_accuracy
#   Minimum percentage of correct answers required to clear an active Needs Review status.
#   Recovery is tracked separately for each direction.
#
# review_recovery_min_answers
#   Minimum number of answers that must be given while recovering from Needs Review.
#   The status is cleared only when BOTH this minimum and the recovery accuracy threshold are met.
#   Example with 80% and 5 answers: 4/5 correct clears review; 3/5 does not.
