# Japanese Vocabulary Trainer — Last frame layout
#
# One config entry = one item in a display line.
# Multiple items can share the same line number.
#
# Syntax:
# line=1; label=Question word; db=question; under=true; gap=28
#
# line       = visual line number
# label      = text shown above/beside the database value
# db         = database field name OR virtual field
# under      = true: value below label / false: label and value side-by-side
# gap        = horizontal space after this item, in pixels
#
# Empty values are ignored. If db is empty, its label is ignored too.
#
# Database fields come from the first row of vocabulary.csv.
# Add new columns to vocabulary.csv and they automatically become available here.
#
# Virtual fields for the Last result:
# question, correct_answer, answer_given, result
#
# Example with three fields on one line and values underneath:
# line=1; label=Question; db=question; under=true; gap=28
# line=1; label=Correct; db=correct_answer; under=true; gap=28
# line=1; label=Given; db=answer_given; under=true; gap=28
#
