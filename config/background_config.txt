# Japanese Vocabulary Trainer — Section-specific background configuration
#
# This file controls the decorative background shown behind the application.
# Configuration is section-specific. A new application section can use a
# background simply by adding a new [section] block; no application-code change
# is required.
#
# Supported sections in the current application:
#   quiz
#   training
#   words
#   backup
#
# [default] is used when a section has no own configuration.
#
# Settings:
# image               = image filename or path. A bare filename is loaded from
#                       backgrounds/. A path containing / is used as written.
# default_hide        = on/off. Initial visibility of the background.
# size                = CSS background-size value, for example cover, contain,
#                       auto, or a CSS length/percentage expression.
# position            = CSS background-position fallback value, for example
#                       center center, right top, or 50% 20%.
# position_portrait   = optional CSS background-position used in portrait
#                       viewport orientation. Falls back to position.
# position_landscape  = optional CSS background-position used in landscape
#                       viewport orientation. Falls back to position.
# background_opacity  = 1–100. Opacity of the background image itself.
# surface_opacity    = 1–100. Opacity of translucent application surfaces.
#                       80 means the surface is 80% opaque, so about 20% of the
#                       background can show through it.
#
# Supported background image formats:
#   JPEG (.jpg, .jpeg), PNG (.png), and WebP (.webp).
# The application uses the browser image loader. Unsupported or invalid image
# files are reported through the central configuration-warning system and use
# the configured fallback behavior.
#
# background_opacity and surface_opacity are independent.
# The background is a separate visual layer and does not participate in layout,
# vocabulary, quiz, training, or other application logic.
#
# If the configured image cannot be loaded, the application keeps working and
# falls back to [default] when available, otherwise to the normal application
# background. The existing configuration-warning indicator reports the problem.
#
# Background visibility is controlled only by default_hide in this file.
# There is no runtime hide/show button in the application UI.
# Orientation means the CSS viewport orientation.

[default]
image=kawaii_01.png
default_hide=on
size=cover
position=center center
background_opacity=60
surface_opacity=80
