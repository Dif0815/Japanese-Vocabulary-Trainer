# Japanese Vocabulary Trainer — Last frame layout
#
# Global layout settings:
# column_gap       = horizontal space between shared columns, in pixels
# row_gap          = vertical space between visual lines, in pixels
# label_font_size  = label font size, in pixels
# label_color      = label text color
#
# Color examples:
# label_color=#777777
# label_color=#999999
# label_color=darkslategray
# label_color=rgb(100,100,100)
#
# The column grid is calculated across ALL visual lines together.
# This means column 1, column 2, etc. stay aligned from one line to the next.
#
# One config entry = one item in a display line.
# Multiple items can share the same line number.
#
# Syntax:
# line=1; label=Question word; db=question; under=true
# `furigana` is an optional presentation parameter for a layout line.
# Example with Furigana enabled:
# line=1; label=Dictionary form; db=japanese; furigana=on; under=true
# Example with Furigana disabled:
# line=1; label=Dictionary form; db=japanese; furigana=off; under=true
# furigana=on reacts to the corresponding section's Show Furigana button.
# furigana=off always displays the raw database value and does not react to the Show Furigana button.
# furigana=on is currently supported only with db=japanese.
# furigana=off may be used with any DB field.
# furigana may be placed anywhere within the same line definition; parameter order does not matter.
#
# line       = visual line number
# label      = text shown above/beside the database value
# db         = database field name OR virtual field
# furigana   = optional presentation attribute for configurable database fields.
#              on reacts to the Quiz Show Furigana button; off stays raw.
#              `furigana=on` is currently supported only for db=japanese.
# under      = true: value below label / false: label and value side-by-side
#
# Empty values are ignored. If db is empty, its label is ignored too.
# If an entire logical line has no visible items, that line consumes no vertical space.
# Later populated lines are automatically compacted upward.
#
# Database fields come from the first row of data/vocabulary.csv.
# Add new columns to data/vocabulary.csv and they automatically become available here.
#
# Virtual fields for the Last result:
# question = the question shown for that Last result.
# correct_answer = the correct answer for that Last result, using the same effective direction as question.
# answer_given = exactly what the learner entered.
# direction = the effective direction belonging to that Last result.
# type_answer_given = learner's actual subtype answer, shown using the configured subtype label.
# reading_if_different = reading only when japanese and reading differ; otherwise empty.
#                       Available in Last and intended to accompany the correct-answer information.
# Mixed mode is resolved to an effective direction for the individual Last result.
# Show Furigana in Quiz controls the Japanese question/correct-answer presentation in Last.
# For configurable db=japanese entries, use furigana=on/off to control whether the field participates.
#
column_gap=28
row_gap=16
label_font_size=12
label_color=#777777

line=1; label=Question word; db=question; under=true
line=1; label=Correct answer; db=correct_answer; under=true
line=1; label=Reading; db=reading_if_different; under=true

line=2; label=Word type; db=type; under=true

line=4; label=Answer given; db=answer_given; under=true
line=4; label=Type answer given; db=type_answer_given; under=true
