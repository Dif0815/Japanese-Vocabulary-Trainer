# Japanese Vocabulary Trainer — Words popup layout
#
# This file controls the popup shown when a word in the Words section is clicked.
# There is ONE customization file for all word types.
#
# A section name must exactly match the value in the CSV type column.
# If a matching section exists, it is used.
# If no matching section exists, [fallback] is used.
# This means a newly introduced word type can be customized simply by adding a new section.
#
# Global layout settings:
# column_gap       = horizontal space between shared columns, in pixels
# row_gap          = vertical space between visual lines, in pixels
# label_font_size  = label font size, in pixels
# label_color      = label text color
#
# Color examples:
# label_color=#777777
# label_color=#999999
# label_color=darkslategray
# label_color=rgb(100,100,100)
#
# The column grid is calculated across ALL visual lines together.
# Column 1, column 2, etc. therefore stay aligned between lines.
#
# Syntax:
# line=1; label=English; db=english; under=true
#
# line  = visual line number
# label = text shown above/beside the value
# db    = CSV field name or supported virtual field
# under = true: value below label / false: label and value side-by-side
#
# Supported virtual fields:
# question          = Japanese dictionary form.
# furigana          = Japanese dictionary form with its CSV reading displayed above it.
# english           = English display (including automatic "to" for verbs)
# type_label        = CSV word type
# verb_type         = verb group as a readable label
# adjective_type    = adjective group as a readable label
#
# Persistent learning-statistic fields are also available to the Word Popup layout.
# These values are stored in browser learning data, not in data/vocabulary.csv.
# They can be used with db= exactly like other popup fields.
#
# ja_asked       = number of Japanese -> English translation questions asked
# ja_correct     = number of Japanese -> English translation questions answered correctly
# ja_wrong       = number of Japanese -> English translation questions answered incorrectly
# en_asked       = number of English -> Japanese translation questions asked
# en_correct     = number of English -> Japanese translation questions answered correctly
# en_wrong       = number of English -> Japanese translation questions answered incorrectly
# last_added     = timestamp when the word was first initialized in learning data
# last_ja_correct = timestamp of the most recent correct Japanese -> English answer
# last_en_correct = timestamp of the most recent correct English -> Japanese answer
# ja_wrong_since_correct = wrong Japanese -> English answers since the last correct answer
# en_wrong_since_correct = wrong English -> Japanese answers since the last correct answer
# ja_review_active = whether Japanese -> English is currently in Needs Review
# ja_review_asked = answers given while recovering Japanese -> English review
# ja_review_correct = correct answers while recovering Japanese -> English review
# en_review_active = whether English -> Japanese is currently in Needs Review
# en_review_asked = answers given while recovering English -> Japanese review
# en_review_correct = correct answers while recovering English -> Japanese review
#
# Any field present in the data/vocabulary.csv header can also be used directly.
# Empty values are ignored, including their labels.
# If an entire logical line has no visible items, that line consumes no vertical space.
# Later populated lines are automatically compacted upward.
#
column_gap=28
row_gap=16
label_font_size=12
label_color=#777777

# ---------------- VERB ----------------
[verb]
line=1; label=English; db=english; under=true
line=2; label=Dictionary form; db=japanese; under=true
