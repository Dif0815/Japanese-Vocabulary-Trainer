# ============================================================
# OPTIONS DEFAULTS
# ============================================================
#
# This file defines the settings used when the application starts.
# It does NOT change data/vocabulary.csv and it does NOT change statistics.
#
# The three main sections use Direction / Vocabulary / Group.
#
# -------------------- DIRECTION -----------------------------
# ja    = Japanese -> English
# en    = English -> Japanese
# mixed = randomly choose Japanese -> English or English -> Japanese
#
# -------------------- FILTER / SELECTION MODE --------------
# native  = use native browser selection controls where applicable.
# custom  = use the custom selection UI.
#
# This setting applies to the Words/Group custom filters and also to
# the Direction selector in Quiz, Training, and Words.
# It does not change the filter data, selection logic or semantics.
#
# ---------------- MENU OPENING DIRECTION -------------------
# filter_menu_direction controls the opening direction of custom selection
# menus in each section:
#   auto = choose based on available viewport space
#   up   = force the menu to open upward
#   down = force the menu to open downward
#
# In native mode the browser/OS controls the native selector's opening
# behavior. filter_menu_direction applies to custom selection menus.
#
# -------------------- VOCABULARY ----------------------------
# all = all currently compatible values selected.
# none = no values selected -> zero results.
# multiselect: value1,value2 = explicitly select several values.
# The values must exist in data/vocabulary.csv.
#
# -------------------- GROUP ---------------------------------
# all = all currently compatible values selected.
# none = no values selected -> zero results.
# empty = select only words whose Group field is empty.
# multiselect: weather,travel = explicitly select several groups.
# The Group filter also accepts (empty) inside a multiselect.
#
# Word Type and Group are multi-select filters. OR applies within one filter;
# AND applies between Vocabulary and Group.
# Each option remains visible and shows its current compatible count.
#
# Invalid filter values and unknown filter_selection_mode values produce a warning
# in the central warning indicator and use the documented fallback.
#
# ---------------- CUSTOM FILTER WIDTH ---------------------
[global]
# Maximum width of custom filter/selection dropdown menus, in pixels.
# The menu minimum width always follows its own control. The menu may expand
# to fit its content up to this global maximum, subject to available viewport width.
# This applies to the existing Words/Group filters and the custom Direction selector.
custom_filter_max_width=420

# -------------------- SHOW FURIGANA -------------------------
# on  = show supported Japanese text with its CSV reading above it when the
#       section's Show Furigana control is ON.
# off = show the raw database value.
#
# Show Furigana is presentation-only. It never selects or generates a new question.
# Quiz and Training can also toggle it by tapping the question itself.
# Words controls the Word Popup presentation too.
#
# In configurable layout files, furigana is an optional presentation attribute:
#     db=japanese;furigana=on
#     db=japanese;furigana=off
# `furigana=on` is currently supported only for db=japanese.
#
# -------------------- ASK TYPE ------------------------------
# Quiz only.
# on  = ask the word type after the translation answer
# off = do not ask the word type
# Changing this during a quiz does NOT advance the current question.
#
# ---------------- QUIZ SKIP KEY ----------------------------
# Quiz only.
# off = no keyboard shortcut triggers Skip.
# ESC = Escape triggers the existing Skip action, including while the
#       Quiz answer field is focused.
# Right Arrow is not used by this feature.
#
# ---------------- CORRECT ANSWER DELAY ---------------------
# Quiz only. correct_answer_delay is the delay in milliseconds after the complete correct-answer flow
# (including Ask Type, when enabled) before the next question is shown.
#
# -------------------- REVEAL ALL ----------------------------
# Training only.
# on  = reveal all configured answers at once
# off = reveal configured answers one step at a time
# Changing this during Training does NOT advance the current word.
#
# -------------------- WORDS SECTION -------------------------
# Direction controls which directional learning/review status is shown:
# mixed = show both Japanese -> English and English -> Japanese
# ja    = show only Japanese -> English
# en    = show only English -> Japanese
# Vocabulary and Group use the same dynamic filters as Quiz and Training.
#
# ============================================================
# QUIZ DEFAULTS
# ============================================================
[quiz]
ask_type=on
direction=mixed
vocabulary=all
group=all
filter_selection_mode=custom
filter_menu_direction=auto
show_furigana=on
quiz_skip_key=ESC
# Delay after a fully correct Quiz answer before the next question is shown, in milliseconds.
correct_answer_delay=800

# ============================================================
# TRAINING DEFAULTS
# ============================================================
[training]
direction=mixed
vocabulary=all
group=all
reveal_all=off
filter_selection_mode=custom
filter_menu_direction=auto
show_furigana=on

# ============================================================
# WORDS DEFAULTS
# ============================================================
[words]
direction=mixed
vocabulary=all
group=all
filter_selection_mode=custom
filter_menu_direction=auto
show_furigana=on
status_filter_learned=on
status_filter_not_learned=on
status_filter_needs_review=on
status_filter_partially_learned=on
